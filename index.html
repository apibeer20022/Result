import matplotlib.pyplot as plt import seaborn as sns import numpy as np # ข้อมูลจากผลการทดลองของคุณเบียร์ (ตัวอย่าง Rest EDV) [cite: 14, 378] qgs_data = np.random.normal(67, 41, 147) # สมมติข้อมูลตาม Median/IQR # 4DM = (1.18 * QGS) + 6.27 fdm_data = (1.18 * qgs_data) + 6.27 + np.random.normal(0, 11.74, 147) def plot_bland_altman(x, y, title): mean = np.mean([x, y], axis=0) diff = x - y md = np.mean(diff) sd = np.std(diff, axis=0) plt.figure(figsize=(8, 5)) plt.scatter(mean, diff, alpha=0.5) plt.axhline(md, color='red', linestyle='--', label=f'Mean Bias: {md:.2f}') plt.axhline(md + 1.96*sd, color='gray', linestyle='--') plt.axhline(md - 1.96*sd, color='gray', linestyle='--') plt.title(f'Bland-Altman Plot: {title}') plt.xlabel('Average of QGS and 4D-M') plt.ylabel('Difference (QGS - 4D-M)') plt.legend() plt.show() # รันเพื่อดูผล plot_bland_altman(qgs_data, fdm_data, "Rest EDV Comparison")
