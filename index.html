import pandas as pd import numpy as np import matplotlib.pyplot as plt import seaborn as sns from scipy import stats # ----------------------------------------- # 1. โหลดข้อมูล (นำเข้าข้อมูลจริงของคุณที่นี่) # ตัวอย่าง: df = pd.read_csv('cardiac_data.csv') # ----------------------------------------- # สร้าง Mock Data ให้คล้ายคลึงกับผลลัพธ์จากบทที่ 4 np.random.seed(42) qgs_edv = np.random.normal(67, 20, 147) # 4DM มักจะสูงกว่า QGS ตามสมการทำนาย: 4DM = (1.18 * QGS) + 6.27 dm4_edv = (1.18 * qgs_edv) + 6.27 + np.random.normal(0, 5, 147) df = pd.DataFrame({'QGS_Rest_EDV': qgs_edv, '4DM_Rest_EDV': dm4_edv}) # ----------------------------------------- # 2. ตั้งค่าธีมสำหรับ Visualization # ----------------------------------------- sns.set_theme(style="whitegrid", palette="muted") plt.rcParams.update({'font.size': 12, 'font.family': 'sans-serif'}) fig, axes = plt.subplots(1, 2, figsize=(16, 6)) # ========================================== # กราฟซ้าย: Regression Scatter Plot (ดูความสัมพันธ์) # ========================================== ax1 = axes[0] sns.regplot(x='QGS_Rest_EDV', y='4DM_Rest_EDV', data=df, scatter_kws={'alpha': 0.6, 'color': '#1f77b4'}, line_kws={'color': '#d62728', 'lw': 2}, ax=ax1) # เพิ่มค่าสถิติลงในกราฟ rho, p_val = stats.spearmanr(df['QGS_Rest_EDV'], df['4DM_Rest_EDV']) textstr = f"Spearman's rho = {rho:.3f}\np < 0.001*" props = dict(boxstyle='round', facecolor='white', alpha=0.9, edgecolor='gray') ax1.text(0.05, 0.95, textstr, transform=ax1.transAxes, fontsize=12, verticalalignment='top', bbox=props) ax1.set_title('Correlation of Rest EDV (QGS vs 4D-M)', fontsize=14, weight='bold') ax1.set_xlabel('QGS Rest EDV (mL)') ax1.set_ylabel('Corridor4DM Rest EDV (mL)') # ========================================== # กราฟขวา: Bland-Altman Plot (ดูความสอดคล้อง) # ========================================== ax2 = axes[1] mean_val = (df['QGS_Rest_EDV'] + df['4DM_Rest_EDV']) / 2 diff_val = df['4DM_Rest_EDV'] - df['QGS_Rest_EDV'] median_bias = np.median(diff_val) lower_loa = np.percentile(diff_val, 2.5) upper_loa = np.percentile(diff_val, 97.5) ax2.scatter(mean_val, diff_val, alpha=0.6, color='#2ca02c') # วาดเส้น Bias และ Limits of Agreement (LoA) ax2.axhline(median_bias, color='blue', linestyle='-', lw=2, label=f'Median Bias: {median_bias:.1f}') ax2.axhline(upper_loa, color='red', linestyle='--', lw=2, label=f'Upper 95% LoA: {upper_loa:.1f}') ax2.axhline(lower_loa, color='red', linestyle='--', lw=2, label=f'Lower 95% LoA: {lower_loa:.1f}') ax2.set_title('Bland-Altman Plot: Rest EDV Difference', fontsize=14, weight='bold') ax2.set_xlabel('Average of QGS and Corridor4DM (mL)') ax2.set_ylabel('Difference (Corridor4DM - QGS) (mL)') ax2.legend(loc='upper right') plt.tight_layout() plt.savefig('spect_comparison_plot.png', dpi=300) plt.show()
